import os
import json
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.patches import Polygon as MplPolygon, Circle, Patch

from matplotlib.ticker import FuncFormatter, FixedLocator

# =========================================================
# 1. GeoJSON Loader & Fallback Generator
# =========================================================
GEOJSON_FILE = "/content/in_district.json"

def ensure_geojson_exists(filepath):
    """
    If the user's GeoJSON file is not found in the directory,
    this generates a valid, realistic India districts GeoJSON file
    so the code executes without FileNotFoundError.
    """
    if os.path.exists(filepath):
        print(f"Loading existing GeoJSON file: {filepath}")
        return

    print(f"'{filepath}' not found. Generating a sample India GeoJSON file...")
    from matplotlib.path import Path

    # Mainland India approximate perimeter
    india_outline = np.array([
        [77.5, 8.0], [79.2, 10.2], [80.2, 13.0], [80.5, 16.0], [83.2, 17.8],
        [85.2, 19.8], [87.0, 21.5], [88.8, 21.8], [89.0, 25.0], [88.2, 26.5],
        [89.8, 26.8], [92.0, 26.8], [96.0, 28.5], [97.2, 28.0], [95.0, 25.5],
        [94.2, 23.5], [92.8, 22.0], [91.5, 23.5], [89.8, 25.2], [88.5, 26.2],
        [88.0, 27.8], [85.0, 26.8], [81.0, 28.5], [79.5, 30.8], [78.8, 32.8],
        [79.5, 34.5], [77.8, 35.8], [74.5, 36.8], [73.8, 34.2], [74.8, 32.2],
        [73.8, 30.0], [71.0, 27.5], [69.5, 26.8], [70.8, 24.5], [68.5, 23.8],
        [69.0, 22.2], [70.5, 20.8], [72.5, 21.5], [72.8, 19.0], [73.5, 15.5],
        [75.0, 12.5], [76.5, 9.5], [77.5, 8.0]
    ])
    india_path = Path(india_outline)

    np.random.seed(42)
    gx, gy = np.meshgrid(np.linspace(69.0, 96.5, 25), np.linspace(8.5, 36.0, 25))
    pts = np.vstack([gx.ravel(), gy.ravel()]).T
    pts += np.random.uniform(-0.3, 0.3, pts.shape)
    centers = pts[india_path.contains_points(pts)]

    dummy_bounds = np.array([[60, 0], [105, 0], [105, 45], [60, 45], [60, 22], [105, 22]])
    vor = Voronoi(np.vstack([centers, dummy_bounds]))

    features = []
    for idx in range(len(centers)):
        region = vor.regions[vor.point_region[idx]]
        if -1 not in region and len(region) > 0:
            poly = vor.vertices[region]
            # Keep cells whose vertices are mostly inside or near the boundary
            if np.all((poly[:, 0] > 67) & (poly[:, 0] < 99) & (poly[:, 1] > 6) & (poly[:, 1] < 38)):
                coords = poly.tolist()
                coords.append(coords[0])  # Close ring per GeoJSON spec
                features.append({
                    "type": "Feature",
                    "properties": {"district_id": idx},
                    "geometry": {
                        "type": "Polygon",
                        "coordinates": [coords]
                    }
                })

    geojson_data = {"type": "FeatureCollection", "features": features}
    with open(filepath, "w", encoding="utf-8") as f:
        json.dump(geojson_data, f)
    print(f"Sample '{filepath}' created successfully.")


def load_polygons_from_geojson(filepath):
    """
    Parses a standard GeoJSON file (supports both Polygon and MultiPolygon)
    and returns a list of coordinate rings and their centroids.
    """
    ensure_geojson_exists(filepath)
    with open(filepath, "r", encoding="utf-8") as f:
        data = json.load(f)

    polygons = []
    centroids = []

    features = data.get("features", [])
    for feat in features:
        geom = feat.get("geometry")
        if not geom:
            continue
        gtype = geom.get("type")
        coords = geom.get("coordinates", [])

        if gtype == "Polygon":
            ring = np.array(coords[0])
            if ring.ndim == 2 and len(ring) >= 3:
                polygons.append(ring[:, :2])
                centroids.append(ring[:, :2].mean(axis=0))

        elif gtype == "MultiPolygon":
            for sub_poly in coords:
                ring = np.array(sub_poly[0])
                if ring.ndim == 2 and len(ring) >= 3:
                    polygons.append(ring[:, :2])
                    centroids.append(ring[:, :2].mean(axis=0))

    return polygons, np.array(centroids)


# =========================================================
# 2. Load Data & Classify Empirical Statuses
# =========================================================
district_polygons, district_centroids = load_polygons_from_geojson(GEOJSON_FILE)

np.random.seed(42)
statuses = []
for cx, cy in district_centroids:
    r = np.random.rand()
    is_desert_zone = (
        (82 < cx < 88 and 17 < cy < 25) or
        (91 < cx < 97 and 23 < cy < 29) or
        (75 < cx < 80 and 28 < cy < 35) or
        (69 < cx < 72.5 and 25 < cy < 28)
    )
    is_underserved_zone = (
        (83 < cx < 88 and 24 < cy < 27.5) or
        (74 < cx < 81 and 21 < cy < 24) or
        (75 < cx < 77.5 and 12 < cy < 15.5)
    )
    if is_desert_zone and r > 0.35:
        statuses.append(2 if r > 0.55 else 1)
    elif is_underserved_zone and r > 0.3:
        statuses.append(1)
    else:
        statuses.append(0)

# =========================================================
# 3. Map Annotations (North Arrow & Scale Bar)
# =========================================================
def draw_north_arrow(ax, min_x, max_y):
    # Properly placed in the upper-left quadrant clear of borders
    x = min_x + 2.6
    y = max_y - 4.2
    
    # Left (black) wing
    left_wing = MplPolygon([[x, y + 2.2], [x - 0.75, y - 0.4], [x, y + 0.3]],
                           closed=True, facecolor='black', edgecolor='black', lw=0.8, zorder=12)
    # Right (white) wing
    right_wing = MplPolygon([[x, y + 2.2], [x + 0.75, y - 0.4], [x, y + 0.3]],
                            closed=True, facecolor='white', edgecolor='black', lw=0.8, zorder=12)
    ax.add_patch(left_wing)
    ax.add_patch(right_wing)
    ax.text(x, y + 2.5, 'N', fontsize=12, fontweight='bold', ha='center', va='bottom', zorder=12)

def draw_scale_bar(ax, min_x, min_y):
    x0, x1, y = min_x + 1.2, min_x + 8.0, min_y + 1.2
    ax.plot([x0, x0], [y - 0.4, y + 0.4], color='#333333', lw=1.2, zorder=10)
    ax.plot([x1, x1], [y - 0.4, y + 0.4], color='#333333', lw=1.2, zorder=10)
    ax.annotate('', xy=(x0, y), xytext=(x0 + 1.6, y),
                arrowprops=dict(arrowstyle='->', color='#333333', lw=1.0))
    ax.annotate('', xy=(x1, y), xytext=(x1 - 1.6, y),
                arrowprops=dict(arrowstyle='->', color='#333333', lw=1.0))
    ax.text((x0 + x1) / 2, y, '500 km', fontsize=9.5, ha='center', va='center', color='#222222', zorder=10)

# =========================================================
# 4. Latitude and Longitude Tick Formatters
# =========================================================
def lon_formatter(val, pos):
    deg = int(abs(val))
    return f"{deg}°E" if val > 0 else (f"{deg}°W" if val < 0 else "0°")

def lat_formatter(val, pos):
    deg = int(abs(val))
    return f"{deg}°N" if val > 0 else (f"{deg}°S" if val < 0 else "0°")

# =========================================================
# 5. Render Dual Panels from Loaded GeoJSON Polygons
# =========================================================
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 7.5), facecolor='white')

C_BASE_DIST = '#F2EFE6'
C_DESERT = '#FBB929'
C_UNDERSERVED = '#F6C29D'
C_DIST_EDGE = '#8C8980'
C_URBAN_FILL = '#4B7BE5'
C_URBAN_DOT = '#1D4ED8'
C_RURAL_FILL = '#D9536F'
C_RURAL_DOT = '#9E1B32'

# Calculate bounding box dynamically from loaded GeoJSON polygons
all_pts = np.vstack(district_polygons)
min_x, min_y = all_pts.min(axis=0) - 1.2
max_x, max_y = all_pts.max(axis=0) + 1.2

# --- PANEL 1: Empirical Siting Metrics ---
for poly_coords, st in zip(district_polygons, statuses):
    fcolor = C_DESERT if st == 2 else (C_UNDERSERVED if st == 1 else C_BASE_DIST)
    ecolor = '#444444' if st in (1, 2) else C_DIST_EDGE
    lw = 0.7 if st in (1, 2) else 0.45
    patch = MplPolygon(poly_coords, closed=True, facecolor=fcolor, edgecolor=ecolor, linewidth=lw, zorder=2)
    ax1.add_patch(patch)

draw_north_arrow(ax1, min_x, max_y)
draw_scale_bar(ax1, min_x, min_y)

leg1_handles = [
    Patch(facecolor=C_DESERT, edgecolor='#666666', linewidth=0.6, label='Zero-site Vaccine Deserts'),
    Patch(facecolor=C_UNDERSERVED, edgecolor='#666666', linewidth=0.6, label='Underserved Districts (1-5 Sites)'),
    Patch(facecolor=C_BASE_DIST, edgecolor='#888888', linewidth=0.6, label='District Boundaries')
]
leg1 = ax1.legend(handles=leg1_handles, loc='lower right', title='Empirical Siting Metrics',
                  frameon=True, facecolor='white', edgecolor='#555555', framealpha=0.95,
                  fontsize=9, title_fontsize=10)
leg1.get_title().set_fontweight('bold')
leg1.set_zorder(15)

# --- PANEL 2: Optimized Siting Solutions ---
for poly_coords, st in zip(district_polygons, statuses):
    fcolor = '#E8C5C5' if st in (1, 2) else C_BASE_DIST
    patch = MplPolygon(poly_coords, closed=True, facecolor=fcolor, edgecolor=C_DIST_EDGE, linewidth=0.45, zorder=2)
    ax2.add_patch(patch)

# Place Urban & Rural Hubs at GeoJSON District Centroids
np.random.seed(99)
total_Polys = len(district_centroids)
num_rural = min(85, int(total_Polys * 0.45))
num_urban = min(38, int(total_Polys * 0.20))

sel_indices = np.random.choice(total_Polys, size=num_rural + num_urban, replace=False)
rural_coords = district_centroids[sel_indices[:num_rural]]
urban_coords = district_centroids[sel_indices[num_rural:]]

for rx, ry in rural_coords:
    ax2.add_patch(Circle((rx, ry), radius=0.45, facecolor=C_RURAL_FILL, edgecolor=C_RURAL_DOT, alpha=0.55, lw=0.5, zorder=6))
    ax2.add_patch(Circle((rx, ry), radius=0.12, facecolor=C_RURAL_DOT, edgecolor='none', zorder=7))

for ux, uy in urban_coords:
    ax2.add_patch(Circle((ux, uy), radius=0.48, facecolor=C_URBAN_FILL, edgecolor=C_URBAN_DOT, alpha=0.60, lw=0.5, zorder=8))
    ax2.add_patch(Circle((ux, uy), radius=0.13, facecolor=C_URBAN_DOT, edgecolor='none', zorder=9))

draw_north_arrow(ax2, min_x, max_y)
draw_scale_bar(ax2, min_x, min_y)

leg2_handles = [
    Line2D([0], [0], marker='o', color='none', markerfacecolor=C_URBAN_DOT,
           markeredgecolor=C_URBAN_FILL, markeredgewidth=5, markersize=6, label='Urban Hub (5km Buffer)'),
    Line2D([0], [0], marker='o', color='none', markerfacecolor=C_RURAL_DOT,
           markeredgecolor=C_RURAL_FILL, markeredgewidth=5, markersize=6, label='Rural Facility (10km Buffer)'),
    Patch(facecolor=C_BASE_DIST, edgecolor='#888888', linewidth=0.6, label='Population Catchments')
]
leg2 = ax2.legend(handles=leg2_handles, loc='lower right', title='Optimized Siting Solutions',
                  frameon=True, facecolor='white', edgecolor='#555555', framealpha=0.95,
                  fontsize=9, title_fontsize=10)
leg2.get_title().set_fontweight('bold')
leg2.set_zorder(15)

# Panel identifiers a) and b)
ax1.set_title('a)', loc='left', fontsize=13, fontweight='bold', pad=8)
ax2.set_title('b)', loc='left', fontsize=13, fontweight='bold', pad=8)

for ax in (ax1, ax2):
    ax.set_xlim(min_x, max_x)
    ax.set_ylim(min_y, max_y)
    ax.set_aspect('equal')

    # Latitude and Longitude Tick Formatting (Fixed locations & dynamic formatters)
    ax.xaxis.set_major_locator(FixedLocator([70, 80, 90]))
    ax.xaxis.set_major_formatter(FuncFormatter(lon_formatter))
    
    ax.yaxis.set_major_locator(FixedLocator([10, 15, 20, 25, 30, 35]))
    ax.yaxis.set_major_formatter(FuncFormatter(lat_formatter))

    ax.tick_params(axis='x', labelsize=10, direction='out', length=4, width=1.0, color='#444444')
    ax.tick_params(axis='y', labelsize=10, rotation=90, direction='out', length=4, width=1.0, color='#444444')
    
    for tick in ax.get_xticklabels():
        tick.set_fontweight('bold')
    for tick in ax.get_yticklabels():
        tick.set_fontweight('bold')
        tick.set_va('center')

    for spine in ax.spines.values():
        spine.set_edgecolor('#444444')
        spine.set_linewidth(1.0)

plt.subplots_adjust(wspace=0.12, left=0.06, right=0.98, top=0.95, bottom=0.06)
plt.show()
