# --- Cartographic Helper Functions ---


def add_north_arrow(ax, x=0.06, y=0.92, length=0.06):
    """Draws a clean, standard academic North arrow inside the map frame."""
    ax.annotate(
        "N",
        xy=(x, y),
        xytext=(x, y - length),
        xycoords="axes fraction",
        textcoords="axes fraction",
        arrowprops=dict(
            facecolor="black",
            edgecolor="black",
            width=1.5,
            headwidth=6.0,
            headlength=7.0,
        ),
        ha="center",
        va="bottom",
        fontsize=9,
        fontweight="bold",
    )


def add_scale_bar(ax, km_dist=500, loc="lower left", pad=0.8):
    """Calibrated scale bar for peninsular India (1° Lon ~ 104.6 km at ~20°N)."""
    deg_len = km_dist / 104.6
    scalebar = AnchoredSizeBar(
        ax.transData,
        deg_len,
        f"{km_dist} km",
        loc=loc,
        pad=pad,
        borderpad=0.8,
        color="black",
        frameon=False,
        size_vertical=0.15,
        fontproperties={"size": 8, "weight": "bold"},
    )
    ax.add_artist(scalebar)
