#Bar Graph

import matplotlib.pyplot as plt
import numpy as np

# Data reconstructed from the graph (ordered bottom to top)
states_data = [
    ("Delhi", 0.11),
    ("Andhra Pradesh", 0.12),
    ("Kerala", 0.22),
    ("Tripura", 0.235),
    ("Haryana", 0.245),
    ("Rajasthan", 0.258),
    ("Uttar Pradesh", 0.260),
    ("Madhya Pradesh", 0.278),
    ("Assam", 0.292),
    ("Bihar", 0.295),
    ("Jammu and Kashmir", 0.305),
    ("Jharkhand", 0.310),
    ("Punjab", 0.325),
    ("Odisha", 0.328),
    ("Mizoram", 0.334),
    ("Gujarat", 0.339),
    ("Himachal Pradesh", 0.342),
    ("Karnataka", 0.345),
    ("West Bengal", 0.348),
    ("Tamil Nadu", 0.358),
    ("Chhattisgarh", 0.390),
    ("Uttarakhand", 0.400),
    ("Meghalaya", 0.404),
    ("Arunachal Pradesh", 0.412),
    ("Maharashtra", 0.450),
    ("Telangana", 0.452),
    ("Nagaland", 0.461),
    ("Manipur", 0.545),
]

states = [item[0] for item in states_data]
values = [item[1] for item in states_data]

# Setup plot
fig, ax = plt.subplots(figsize=(8, 10), dpi=150)

# Generate gradient colors (purple at the bottom to golden-orange at the top)
cmap = plt.cm.get_cmap("inferno")
# Sample the colormap between 0.25 (purple) and 0.85 (orange/yellow)
colors = [cmap(val) for val in np.linspace(0.25, 0.85, len(states))]

# Plot horizontal bars
bars = ax.barh(states, values, color=colors, height=0.75, edgecolor="none")

# National Benchmark vertical line (0.449 from the Gini index in plot 1)
national_benchmark = 0.449
ax.axvline(
    x=national_benchmark,
    color="#d9534f",
    linestyle="--",
    linewidth=1.2,
    label=f"National Benchmark Gini ({national_benchmark:.3f})",
)

# Axis labels and limits
ax.set_xlabel("Intra-State District Gini Disparity Index", fontsize=10, labelpad=8)
ax.set_xlim(0.0, 0.57)
ax.set_ylim(-0.8, len(states) - 0.2)

# Styling and grid
ax.grid(axis="x", linestyle="-", color="#e5e5e5", zorder=0)
ax.grid(axis="y", linestyle="-", color="#f0f0f0", zorder=0)
ax.set_axisbelow(True)

# Tick formatting
ax.tick_params(axis="y", labelsize=8)
ax.tick_params(axis="x", labelsize=9)

# Clean up spines (borders)
for spine in ["top", "right", "left", "bottom"]:
    ax.spines[spine].set_color("#cccccc")

# Legend
ax.legend(loc="lower right", frameon=True, facecolor="white", edgecolor="#cccccc", fontsize=9)

plt.tight_layout()
plt.show()
