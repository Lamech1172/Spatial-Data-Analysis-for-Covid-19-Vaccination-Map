# 1. ROBUST NAME CLEANER & COMPREHENSIVE ALIAS LOOKUP
# ==============================================================================
def clean_str(s):
    """Normalize string: strip punctuation, special tokens, extra spaces, and casing."""
    if not isinstance(s, str):
        return ""
    s = unicodedata.normalize("NFKD", s).encode("ASCII", "ignore").decode("utf-8")
    s = s.lower()
    s = re.sub(r"[\.,\(\)/\-_\*&]", " ", s)
    s = re.sub(r"\b(district|dist|corporation)\b", "", s)
    return re.sub(r"\s+", " ", s).strip()

# 2. INGESTION & DATASET HARMONIZATION
# ==============================================================================
geojson_path = "/content/in_district.json"
csv_path = "cowin_vaccine_data_districtwise.csv"

# Load GeoJSON
gdf = gpd.read_file(geojson_path)
if gdf.crs is None:
    gdf.set_crs(epsg=4326, inplace=True)
else:
    gdf = gdf.to_crs(epsg=4326)

# Locate District column
dist_col = None
for col in ["district", "dtname", "District", "NAME_2", "DISTRICT", "dt_name"]:
    if col in gdf.columns:
        dist_col = col
        break

gdf["clean_dist"] = gdf[dist_col].apply(clean_str)
gdf["clean_dist"] = gdf["clean_dist"].replace(DISTRICT_CORRECTIONS)

# Parse CoWIN CSV
df_raw = pd.read_csv(csv_path, header=None, low_memory=False)
metrics_row = df_raw.iloc[1].values
data_rows = df_raw.iloc[2:].copy()

district_idx = 5
state_idx = 2

dose1_cols = [
    i for i, m in enumerate(metrics_row)
    if "first dose" in str(m).lower()
    and pd.to_numeric(data_rows[i], errors="coerce").fillna(0).max() > 0
]
dose2_cols = [
    i for i, m in enumerate(metrics_row)
    if "second dose" in str(m).lower()
    and pd.to_numeric(data_rows[i], errors="coerce").fillna(0).max() > 0
]

latest_d1 = dose1_cols[-1]
latest_d2 = dose2_cols[-1]

cowin_df = pd.DataFrame({
    "State": data_rows[state_idx],
    "District": data_rows[district_idx],
    "Dose1": pd.to_numeric(data_rows[latest_d1], errors="coerce").fillna(0),
    "Dose2": pd.to_numeric(data_rows[latest_d2], errors="coerce").fillna(0),
}).dropna(subset=["District"])

cowin_df["clean_dist"] = cowin_df["District"].apply(clean_str)
cowin_df["clean_dist"] = cowin_df["clean_dist"].replace(DISTRICT_CORRECTIONS)

# Group duplicate sub-units and sum doses
cowin_agg = cowin_df.groupby("clean_dist", as_index=False)[["Dose1", "Dose2"]].sum()
cowin_agg["dose1_thousands"] = cowin_agg["Dose1"] / 1000.0
cowin_agg["dose2_thousands"] = cowin_agg["Dose2"] / 1000.0

# Merge datasets
target_gdf = gdf.merge(cowin_agg, on="clean_dist", how="left")

# Median fallback for missing minor islands/taluks to prevent grey holes
target_gdf["dose1_thousands"] = target_gdf["dose1_thousands"].fillna(
    target_gdf["dose1_thousands"].median()
)
target_gdf["dose2_thousands"] = target_gdf["dose2_thousands"].fillna(
    target_gdf["dose2_thousands"].median()
)

# Unify outer boundary geometry
india_boundary = target_gdf.unary_union

# 3. CARTOGRAPHIC FORMATTING (FIXED (a) / (b) & LABELS)
# ==============================================================================
def format_cartographic_axes(ax, panel_label, full_title=None):
    """Aligns (a) and (b) cleanly above the top-left edge of the box and formats ticks."""
    # 1. Place (a) / (b) aligned with the left edge of the bounding frame
    ax.set_title(
        f"({panel_label})" if full_title is None else f"({panel_label}) {full_title}",
        loc="left",
        fontsize=12,
        fontweight="bold",
        pad=10,  # Neatly spaced just above the frame line
    )

    # 2. Strict coordinate limits
    ax.set_xlim(67.0, 98.5)
    ax.set_ylim(6.5, 38.0)

    # 3. Suppress stray "x" and "y" axis labels completely
    ax.set_xlabel("")
    ax.set_ylabel("")
    ax.xaxis.label.set_visible(False)
    ax.yaxis.label.set_visible(False)

    # 4. Longitude and Latitude ticks formatting
    ax.set_xticks([70, 80, 90])
    ax.set_yticks([10, 15, 20, 25, 30, 35])
    ax.xaxis.set_major_formatter(
        plt.FuncFormatter(
            lambda v, _: f"{int(abs(v))}°{'E' if v >= 0 else 'W'}"
        )
    )
    ax.yaxis.set_major_formatter(
        plt.FuncFormatter(
            lambda v, _: f"{int(abs(v))}°{'N' if v >= 0 else 'S'}"
        )
    )
    ax.tick_params(direction="out", length=4, width=0.8, labelsize=8)

    for spine in ax.spines.values():
        spine.set_linewidth(0.9)
        spine.set_color("black")
