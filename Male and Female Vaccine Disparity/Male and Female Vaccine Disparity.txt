import matplotlib.colors as mcolors
import matplotlib.pyplot as plt
import matplotlib.ticker as mticker
import numpy as np
import pandas as pd


# Helper functions
def add_north_arrow(ax):
  ax.annotate(
      '▲\nN',
      xy=(0.95, 0.92),
      xycoords='axes fraction',
      fontsize=13,
      fontweight='bold',
      ha='center',
      va='center',
  )


def add_scale_bar(ax, length_km=500):
  ax.annotate(
      f'|←─ {length_km} km ─→|',
      xy=(0.06, 0.08),
      xycoords='axes fraction',
      fontsize=9,
      fontweight='bold',
      bbox=dict(boxstyle='square,pad=0.25', fc='white', ec='none', alpha=0.7),
  )


def lat_formatter(x, pos):
  direction = 'N' if x > 0 else ('S' if x < 0 else '')
  return f'{abs(x):g}°{direction}'


def lon_formatter(x, pos):
  direction = 'E' if x > 0 else ('W' if x < 0 else '')
  return f'{abs(x):g}°{direction}'

        'aspect': 28,
    },

ax.set_xlabel('')
ax.set_ylabel('')

# --- Clean Coordinates & De-clutter Ticks ---
ax.grid(False)

ax.xaxis.set_major_locator(mticker.MultipleLocator(4))
ax.yaxis.set_major_locator(mticker.MultipleLocator(4))

ax.yaxis.set_major_formatter(mticker.FuncFormatter(lat_formatter))

ax.tick_params(
    axis='both',
    which='both',
    direction='out',
    length=4,
    labelsize=8.5,
    colors='#222222',
)

  spine.set_linewidth(0.8)
  spine.set_edgecolor('#333333')

# Elements placement
add_north_arrow(ax)
