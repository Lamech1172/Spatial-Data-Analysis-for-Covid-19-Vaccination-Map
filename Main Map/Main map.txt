def clean_token(text):
    if text is None or pd.isna(text):
        return ""
    clean = str(text).lower().replace("&", "and").replace("*", "").replace(".", "")
    clean = re.sub(r"[^a-z0-9]", "", clean)
    return clean

# ----------------------------------------------------------------------
# 3. Load & Process Datasets
# ----------------------------------------------------------------------
raw_df = pd.read_csv("cowin_vaccine_data_districtwise.csv", low_memory=False)
target_date = "23/08/2021"

data_df = raw_df.iloc[1:][["State", "District", target_date]].copy()
data_df.rename(columns={target_date: "total_doses"}, inplace=True)
data_df["total_doses"] = pd.to_numeric(data_df["total_doses"], errors="coerce").fillna(0)

gdf = gpd.read_file("/content/in_district.json")
if gdf.crs != "EPSG:4326":
    gdf = gdf.to_crs(epsg=4326)

state_col = [c for c in gdf.columns if c.lower() in ["stname", "state_name", "statename", "state"]][0]
dist_col = [c for c in gdf.columns if c.lower() in ["dtname", "district_name", "districtname", "district"]][0]

gdf["state_norm"] = gdf[state_col].apply(lambda x: STATE_ALIAS.get(clean_token(x), clean_token(x)))
gdf["dist_norm"] = gdf[dist_col].apply(clean_token)

data_df["state_norm"] = data_df["State"].apply(lambda x: STATE_ALIAS.get(clean_token(x), clean_token(x)))
data_df["dist_norm"] = data_df["District"].apply(clean_token)

# ----------------------------------------------------------------------
# 4. Bi-Directional Robust Matching
# ----------------------------------------------------------------------
# Map CoWIN records into GeoDataFrame
state_to_geo = defaultdict(list)
for idx, row in gdf.iterrows():
    state_to_geo[row["state_norm"]].append((row["dist_norm"], idx))

# Also allow Telangana <-> Andhra Pradesh cross-lookup if needed
for dist, idx in state_to_geo.get("andhrapradesh", []):
    state_to_geo["telangana"].append((dist, idx))

data_df["geo_idx"] = None
matched_indices = []

for idx_d, row in data_df.iterrows():
    sn, dn = row["state_norm"], row["dist_norm"]
    target_idx = None

    candidates = state_to_geo.get(sn, [])
    cand_names = [c[0] for c in candidates]

    # Step 1: Alias lookup
    override = DISTRICT_ALIAS.get((sn, dn))
    if override:
        for name, g_idx in candidates:
            if name == override or name.startswith(override[:8]):
                target_idx = g_idx
                break

    # Step 2: Exact match
    if target_idx is None:
        for name, g_idx in candidates:
            if name == dn:
                target_idx = g_idx
                break

    # Step 3: Prefix / Substring match (>= 4 chars)
    if target_idx is None:
        for name, g_idx in candidates:
            if len(name) >= 4 and len(dn) >= 4 and (name.startswith(dn) or dn.startswith(name)):
                target_idx = g_idx
                break

    # Step 4: Fuzzy Levenshtein match within state
    if target_idx is None and cand_names:
        matches = difflib.get_close_matches(dn, cand_names, n=1, cutoff=0.58)
        if matches:
            target_idx = [g_idx for name, g_idx in candidates if name == matches[0]][0]

    matched_indices.append(target_idx)

data_df["geo_idx"] = matched_indices

# Aggregate matched values
agg_df = data_df[data_df["geo_idx"].notna()].groupby("geo_idx")["total_doses"].sum().reset_index()

gdf["geo_idx"] = gdf.index
merged_gdf = gdf.merge(agg_df, on="geo_idx", how="left")

# ----------------------------------------------------------------------
# 5. Fix Residual Unmatched Polygons (Parent / Neighbor Imputation)
# ----------------------------------------------------------------------
# Fill any remaining unmatched polygon with the average of its immediate state neighbors
unmatched = merged_gdf["total_doses"].isna() | (merged_gdf["total_doses"] <= 0)

if unmatched.any():
    for g_idx in merged_gdf[unmatched].index:
        st = merged_gdf.loc[g_idx, "state_norm"]
        # Match with any CoWIN data row from the same state that had no polygon
        unclaimed = data_df[(data_df["state_norm"] == st) & (data_df["geo_idx"].isna())]
        if not unclaimed.empty:
            merged_gdf.loc[g_idx, "total_doses"] = unclaimed["total_doses"].values[0]
        else:
            # Impute median of the state so the polygon does not drop out
            state_median = merged_gdf[merged_gdf["state_norm"] == st]["total_doses"].median()
            merged_gdf.loc[g_idx, "total_doses"] = state_median if pd.notna(state_median) and state_median > 0 else 100000

# ----------------------------------------------------------------------
# 6. Classification & Visual Mapping
# ----------------------------------------------------------------------
palette = ["#ffffcc", "#c7e9b4", "#7fcdbb", "#41b6c4", "#1d91c0", "#225ea8", "#0c2c84"]
valid_values = merged_gdf["total_doses"]

# Compute 7 Fisher-Jenks breaks
fj = mapclassify.FisherJenks(valid_values, k=7)
bins = [0] + list(fj.bins)

legend_handles = []
merged_gdf["fill_color"] = palette[0]

for i in range(len(bins) - 1):
    low, high = int(bins[i] + 1), int(bins[i + 1])
    mask = (merged_gdf["total_doses"] >= low) & (merged_gdf["total_doses"] <= high)
    merged_gdf.loc[mask, "fill_color"] = palette[i]
    lbl = f"{low:,} – {high:,}"
    legend_handles.append(Patch(facecolor=palette[i], edgecolor="#333333", linewidth=0.4, label=lbl))
