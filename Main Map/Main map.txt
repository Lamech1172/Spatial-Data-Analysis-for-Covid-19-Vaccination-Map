def _clean_token(text):
    if text is None or pd.isna(text):
        return "", ""
    clean = str(text).lower().replace("&", "and").replace("*", "").replace(".", "")
    clean = re.sub(r"[^a-z0-9\s]", " ", clean)
    tokens = sorted(clean.split())
    compact = re.sub(r"\s+", "", clean)
    sorted_phrase = ""
    if tokens:
        sorted_phrase = "".join(tokens)
    return compact, sorted_phrase

# 2. Load and Process CoWIN Dataset

csv_path = "cowin_vaccine_data_districtwise.csv"
raw_df = pd.read_csv(csv_path, low_memory=False)

target_date = "23/08/2021"
data_df = raw_df.iloc[1:][["State", "District", "District_Key", target_date]].copy()
data_df.rename(columns={target_date: "total_doses"}, inplace=True)
data_df["total_doses"] = pd.to_numeric(data_df["total_doses"], errors="coerce")

# ----------------------------------------------------------------------
# 3. Load District Shapefile / GeoJSON and Match
# ----------------------------------------------------------------------
geoj_path = "/content/in_district.json"
gdf = gpd.read_file(geoj_path)
if gdf.crs != "EPSG:4326":
    gdf = gdf.to_crs(epsg=4326)

state_col = [c for c in gdf.columns if c.lower() in ["stname", "state_name", "statename", "state"]][0]
dist_col = [c for c in gdf.columns if c.lower() in ["dtname", "district_name", "districtname", "district"]][0]

# Standardize spatial datasets for high-fidelity matching
gdf["state_norm"] = gdf[state_col].apply(lambda x: STATE_ALIAS.get(_clean_token(x)[0], _clean_token(x)[0]))
gdf[["dist_norm", "dist_sorted"]] = gdf[dist_col].apply(lambda x: pd.Series(_clean_token(x)))

data_df["state_norm"] = data_df["State"].apply(lambda x: STATE_ALIAS.get(_clean_token(x)[0], _clean_token(x)[0]))
data_df[["dist_norm", "dist_sorted"]] = data_df["District"].apply(lambda x: pd.Series(_clean_token(x)))

# Map indices
state_to_geo = defaultdict(list)
for idx, row in gdf.iterrows():
    state_to_geo[row["state_norm"]].append((row["dist_norm"], row["dist_sorted"], idx))

matched_idx = []
for _, row in data_df.iterrows():
    sn, dn = row["state_norm"], row["dist_norm"]

    # Specialized overrides
    if "sasnagar" in dn or dn == "sas":
        if sn == "punjab":
            punjab_candidates = state_to_geo.get("punjab", [])
            h = [idx for (n, _, idx) in punjab_candidates if "sas" in n or "sasnagar" in n]
            if h:
                matched_idx.append(h[0])
                continue

    # Delhi Specific Outliers
    if sn == "nctofdelhi":
        delhi_candidates = state_to_geo.get("nctofdelhi", [])
        if "southeast" in dn:
            h = [idx for (n, _, idx) in delhi_candidates if "southeast" in n]
            if h: matched_idx.append(h[0]); continue
        elif "southwest" in dn:
            h = [idx for (n, _, idx) in delhi_candidates if "southwest" in n]
            if h: matched_idx.append(h[0]); continue
        elif "south" in dn:
            h = [idx for (n, _, idx) in delhi_candidates if n == "south"]
            if h: matched_idx.append(h[0]); continue

    # Dadra / Nagar Haveli / Daman / Diu
    if "dadra" in dn or ("nagar" in dn and "haveli" in dn):
        dnh_geo = gdf[gdf["dist_norm"].str.contains("dadra|haveli", regex=True, na=False)]
        if not dnh_geo.empty:
            matched_idx.append(dnh_geo.index[0])
            continue

    candidates = state_to_geo.get(sn, [])
    override = DISTRICT_ALIAS.get((sn, dn))
    if override:
        h = [idx for (n, _, idx) in candidates if n == override or n.startswith(override[:12])]
        if h:
            matched_idx.append(h[0])
            continue

    h = [idx for (n, _, idx) in candidates if n == dn]
    if h:
        matched_idx.append(h[0])
        continue

    # Fallback to prefix matching
    h = [idx for (n, _, idx) in candidates if len(n) >= 4 and (dn.startswith(n) or n.startswith(dn))]
    if h:
        matched_idx.append(h[0])
        continue

    if candidates:
        cand_names = [n for (n, _, _) in candidates]
        best = difflib.get_close_matches(dn, cand_names, n=1, cutoff=0.55)
        if best:
            matched_idx.append([idx for (n, _, idx) in candidates if n == best[0]][0])
            continue

    matched_idx.append(None)

data_df["geo_idx"] = matched_idx

# Aggregate and assign matched values back onto spatial geometry dataframe
agg_df = data_df[data_df["geo_idx"].notna()].groupby("geo_idx")["total_doses"].sum().reset_index()
gdf["geo_idx"] = gdf.index
merged_gdf = gdf.merge(agg_df, on="geo_idx", how="left")
merged_gdf["total_doses"] = pd.to_numeric(merged_gdf["total_doses"], errors="coerce").fillna(0)

    if pd.isna(val) or val <= 0:
        return no_data_color
    for idx, (low, high) in enumerate(bin_cutoffs):
        if low <= val <= high:
            return colors[idx]
    if val > bin_cutoffs[-1][1]:
        return colors[-1]
    return no_data_color
